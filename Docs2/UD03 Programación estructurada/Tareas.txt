Sentencias CONTINUE y BREAK en condicionales dentro de bucles

while (x < 5) {
   x = x + 1;
   if (x == 3) continue;
   System.out.println(x); 
}  

while (x < 5) {
   x = x + 1; 
   if (x == 3) break;
   System.out.println(x); 
}
